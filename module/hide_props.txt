# Global system-property spoofing, applied once at boot through `resetprop`.
#
#   - `name=value` sets the system property `name` to `value` globally, so every process on the
#     device sees the spoofed value. This is applied at boot only; edit and reboot to change it.
#   - Lines starting with `#` are comments and blank lines are ignored.
#   - Properties whose value already matches are skipped, so nothing is written unnecessarily.
#   - Properties this build never defined are never created: inserting a new property entry both
#     moves the property-area mtime and leaves a non-AOSP property behind, which tamper checks flag
#     (e.g. `sys.oem_unlock_allowed` is only defined up to Android 15). Such lines are no-ops.
#   - Remove a line to stop spoofing that property (the original value returns on the next boot).
#
#   - An `@<package>` line opens a foreground-only scope: the lines under it are applied *only*
#     while that package is the foreground activity, and reverted the moment it leaves. Use it to
#     satisfy a detector that insists on a specific value (e.g. `persist.sys.usb.config=mtp`) without
#     actually keeping that behaviour on (the global `usb=none` stays in force everywhere else).
#     The original value of every property in the scope is snapshotted and restored on exit, so the
#     override never leaks outside the detector's foreground window.
#
# Behaviour switches (keys without a dot):
#   adb=off    disable USB debugging at boot (Settings.Global.adb_enabled = 0)
#   usb=none   set the USB configuration to `none`, i.e. no data functions (MTP/ADB/…)

# --- Verified boot / bootloader lock state ---
ro.boot.verifiedbootstate=green
ro.boot.flash.locked=1
ro.boot.vbmeta.device_state=locked
ro.boot.veritymode=enforcing
ro.boot.warranty_bit=0
ro.warranty_bit=0
ro.vendor.boot.warranty_bit=0
ro.vendor.warranty_bit=0
sys.oem_unlock_allowed=0
ro.oem_unlock_supported=0

# --- Build / debug state ---
ro.secure=1
ro.debuggable=0
ro.build.type=user
ro.build.tags=release-keys
ro.boot.selinux=enforcing

# --- Boot hardening ---
# USB data stays off globally (`none`); the detector-only override lives in the scope below.
adb=off
usb=none

# --- Foreground-only overrides ---
# While the listed package is the foreground activity, these properties read their "normal" value
# so the detector's consistency check passes; they revert to the global values the instant it closes.
# `persist.sys.usb.config` / `sys.usb.config` already exist on this build, so the override rewrites
# them in place (no new entry, no property-area mtime change) and USB data is never actually enabled.
@com.studio.duckdetector
persist.sys.usb.config=mtp
sys.usb.config=mtp
