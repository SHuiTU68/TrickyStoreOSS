# Global system-property spoofing, applied once at boot through `resetprop`.
#
# Two forms:
#   name=value   rewrite an existing property in place
#   name         delete the property from the area entirely
#
# Rules the applier follows:
#   - `name=value` sets the property globally, so every process sees the spoofed value. Applied at
#     boot only; edit and reboot to change it.
#   - Lines starting with `#` are comments and blank lines are ignored.
#   - Properties whose value already matches are skipped, so nothing is written unnecessarily.
#   - Properties this build never defined are never created: inserting a new property entry both
#     moves the property-area mtime and leaves a non-AOSP property behind, which tamper checks flag.
#     Such lines are no-ops.
#   - Remove a line to stop spoofing that property (the original value returns on the next boot).
#
# Behaviour switches (keys without a dot):
#   adb=off    disable USB debugging at boot (Settings.Global.adb_enabled = 0)
#   usb=mtp    rewrite the USB configuration properties without notifying property_service
#   fp=on      rebuild every `ro.*.build.fingerprint` from the properties set in this file

# --- Verified boot / bootloader lock state ---
ro.boot.verifiedbootstate=green
ro.boot.flash.locked=1
ro.boot.vbmeta.device_state=locked
ro.boot.veritymode=enforcing
ro.boot.warranty_bit=0
ro.warranty_bit=0
ro.vendor.boot.warranty_bit=0
ro.vendor.warranty_bit=0
ro.oem_unlock_supported=0

# Android 16 (SDK 36) dropped the platform write path for this key, so a build that still exposes it
# was written by something outside the platform. Detection therefore keys off *existence*, not value:
# any non-empty value is treated as tampering, and `0` is not a safe answer. Deleting it is the only
# state that cannot be flagged, and it also drops the key from the observed-rule count.
sys.oem_unlock_allowed

# --- Build / debug state ---
ro.secure=1
ro.debuggable=0
ro.build.type=user
ro.build.tags=release-keys
ro.boot.selinux=enforcing

# --- Fingerprint chain ---
# A fingerprint is structured, not free-form: `brand/product/device:release/id/incremental:type/tags`.
# Every segment is cross-checkable against the `ro.product.<partition>.*` and `ro.build.*` properties
# it claims to describe, so hard-coding some other device's string would *create* the mismatch it is
# supposed to hide. `fp=on` therefore composes the value at boot from the properties this file sets,
# which makes the chain self-consistent by construction, and rewrites only the fingerprints this build
# already defines (`ro.build`, `ro.system.build`, `ro.system_ext.build`, `ro.vendor.build`,
# `ro.odm.build`, `ro.product.build`, `ro.bootimage.build`). When the composition matches what the
# device already reports nothing is written, so this stays a no-op unless the identity above changes.
fp=on

# --- Boot hardening ---
adb=off
# Left commented out on purpose: rewriting sys.usb.config while the gadget keeps its real
# configuration makes the reported state disagree with the bound functions, which is itself a signal.
# Enable only if you also want USB data off for real.
